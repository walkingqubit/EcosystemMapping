    function nodeMaterial(n, highlight=false) {

        shininess: 60,

        transparent: n.type==='TEAM',

        opacity: n.type==='TEAM' ? 0.85 : 1.0

    function nodeGeometry(n, size) {

        case 'APPLICATION': return new THREE.SphereGeometry(size, 24, 18);

        case 'CLOUD_SERVICE': return new THREE.CylinderGeometry(size*0.7, size*0.7, size*2.1, 18);

        case 'GATEWAY': return new THREE.ConeGeometry(size*0.9, size*2.0, 18);

        case 'TEAM': return new THREE.TorusGeometry(size*1.1, size*0.35, 12, 32);

      }

      return new THREE.SphereGeometry(size, 16, 12);

    }


    function getNodeSize(n) {

      return Math.max(2.5, Math.min(7.5, 2.5 + deg*0.7));

    }


    function createNodeMesh(n) {

      const size = getNodeSize(n);

      const geom = nodeGeometry(n, size);

      const mat = nodeMaterial(n);

      const mesh = new THREE.Mesh(geom, mat);

      mesh.position.copy(nodePos[n.id]);

      mesh.userData = {node: n};

      mesh.castShadow = true;

      mesh.receiveShadow = true;

      return mesh;

    }


    function createEdgeMesh(src, tgt, idx) {

      // Animated dashed line

      const points = [nodePos[src].clone(), nodePos[tgt].clone()];

      const curve = new THREE.CatmullRomCurve3(points);

      const geom = new THREE.TubeGeometry(curve, 16, 0.45, 8, false);

      const mat = new THREE.MeshBasicMaterial({

        color: 0xffffff,

        transparent: true,

        opacity: 0.45,

        wireframe: false

      });

      const mesh = new THREE.Mesh(geom, mat);

      mesh.userData = {src, tgt, idx};

      return mesh;

    }


    function createAnimatedDashLine(src, tgt, idx) {

      // Use Line2 for dashes, fallback to animated opacity

      const mat = new THREE.LineDashedMaterial({

        color: 0xffffff,

        dashSize: 3,

        gapSize: 2,

        linewidth: 2,

        transparent: true,

        opacity: 0.7

      });

      const geom = new THREE.BufferGeometry().setFromPoints([

        nodePos[src].clone(), nodePos[tgt].clone()

      ]);

      const line = new THREE.Line(geom, mat);

      line.computeLineDistances();

      line.userData = {src, tgt, idx};

      return line;

    }


    function createStarfield() {

      const starGeo = new THREE.BufferGeometry();

      const starCount = 1200;

      const positions = [];

      for (let i=0; i<starCount; ++i) {

        const r = 300 + Math.random()*200;

        const theta = Math.random()*2*Math.PI;

        const phi = Math.acos(2*Math.random()-1);

        positions.push(

          Math.sin(phi)*Math.cos(theta)*r,

          Math.sin(phi)*Math.sin(theta)*r,

          Math.cos(phi)*r

        );

      }

      starGeo.setAttribute('position', new THREE.Float32BufferAttribute(positions,3));

      const starMat = new THREE.PointsMaterial({color:0xffffff, size:1.2, sizeAttenuation:true, transparent:true, opacity:0.7});

      return new THREE.Points(starGeo, starMat);

    }


    function buildScene() {

      scene = new THREE.Scene();

      scene.background = new THREE.Color(0x11131a);

      scene.add(createStarfield());


      // Lighting

      const amb = new THREE.AmbientLight(0xffffff, 0.38);

      scene.add(amb);

      const dir = new THREE.DirectionalLight(0xffffff, 0.65);

      dir.position.set(80,120,60);

      dir.castShadow = true;

      scene.add(dir);


      // Nodes

      nodeMeshes = {};

      nodes.forEach(n => {

        if (!visibleTypes.has(n.type)) return;

        const mesh = createNodeMesh(n);

        scene.add(mesh);

        nodeMeshes[n.id] = mesh;

      });


      // Edges

      edgeMeshes = [];

      edgeAnimOffsets = [];

      edges.forEach((e,idx) => {

        if (!nodeMeshes[e.source] || !nodeMeshes[e.target]) return;

        const line = createAnimatedDashLine(e.source, e.target, idx);

        scene.add(line);

        edgeMeshes.push(line);

        edgeAnimOffsets.push(Math.random()*10);

      });

    }


    function updateNodeVisibility() {

      Object.values(nodeMeshes).forEach(mesh => {

        const n = mesh.userData.node;

        mesh.visible = visibleTypes.has(n.type);

      });

      edgeMeshes.forEach(line => {

        const src = line.userData.src, tgt = line.userData.tgt;

        line.visible = nodeMeshes[src]?.visible && nodeMeshes[tgt]?.visible;

      });

    }


    function updateNodeMaterials() {

      Object.values(nodeMeshes).forEach(mesh => {

        const n = mesh.userData.node;

        mesh.material = nodeMaterial(n, mesh.userData.highlight);

      });

    }


    function highlightNode(id, highlight=true) {

      Object.values(nodeMeshes).forEach(mesh => mesh.userData.highlight = false);

      if (id && nodeMeshes[id]) nodeMeshes[id].userData.highlight = highlight;

      updateNodeMaterials();

    }

      edgeMeshes.forEach((line,idx) => {

        // Animate dash offset

        line.material.dashOffset = -0.03*(time/16 + edgeAnimOffsets[idx]);

        line.material.needsUpdate = true;

      });

    }


    function resize() {

      const w = window.innerWidth, h = window.innerHeight;

      camera.aspect = w/h;

      camera.updateProjectionMatrix();

      renderer.setSize(w,h);

    }


    function flyToNode(id) {

      if (!nodeMeshes[id]) return;

      const pos = nodeMeshes[id].position.clone();

      const camTarget = pos.clone().add(new THREE.Vector3(0,8,18));

      // Animate camera

      let t = 0, start = camera.position.clone(), end = camTarget;

      let anim = () => {

        t += 0.04;

        camera.position.lerpVectors(start, end, Math.min(1,t));

        controls.target.lerpVectors(controls.target, pos, Math.min(1,t));

        controls.update();

        if (t<1) requestAnimationFrame(anim);

      };

      anim();

    }


    // --- Raycasting for hover/click ---

    function getIntersectedNode(mx,my) {

      const rect = renderer.domElement.getBoundingClientRect();

      mouse.x = ((mx-rect.left)/rect.width)*2-1;

      mouse.y = -((my-rect.top)/rect.height)*2+1;

      raycaster.setFromCamera(mouse, camera);

      const intersects = raycaster.intersectObjects(Object.values(nodeMeshes).filter(m=>m.visible), false);

      if (intersects.length>0) return intersects[0].object.userData.node;

      return null;

    }

    // --- Decommission Mode ---

    function computeDecomImpact(mainframeId) {

      // BFS from mainframe, collect all dependent nodes

      const impacted = new Set();

      const queue = [mainframeId];

      while (queue.length) {

        const curr = queue.shift();

        edges.forEach(e => {

          if (e.source===curr && !impacted.has(e.target)) {

            impacted.add(e.target);

            queue.push(e.target);

          }

        });

      }

      return impacted;

    }

    function main() {

      // Renderer

      renderer = new THREE.WebGLRenderer({antialias:true});

      renderer.setClearColor(0x11131a);

      renderer.setSize(window.innerWidth, window.innerHeight);

      renderer.shadowMap.enabled = true;

      document.body.appendChild(renderer.domElement);


      // Camera & Controls

      camera = new THREE.PerspectiveCamera(60, window.innerWidth/window.innerHeight, 0.1, 2000);

      camera.position.set(0, 60, 120);

      controls = new THREE.OrbitControls(camera, renderer.domElement);

      controls.enableDamping = true;

      controls.dampingFactor = 0.13;

      controls.minDistance = 30;

      controls.maxDistance = 400;

      controls.target.set(0,0,0);


      raycaster = new THREE.Raycaster();

      mouse = new THREE.Vector2();


      buildScene();


      // Events

      window.addEventListener('resize', resize);

      renderer.domElement.addEventListener('mousemove', onMouseMove);

      renderer.domElement.addEventListener('click', onMouseClick);


      // Filter checkboxes

      document.querySelectorAll('.filter-checkbox').forEach(cb => {

        cb.addEventListener('change', e => {

          visibleTypes = new Set(Array.from(document.querySelectorAll('.filter-checkbox')).filter(c=>c.checked).map(c=>c.dataset.type));

          buildScene();

          updateNodeVisibility();

        });

      });


      // Search

      document.getElementById('search-btn').onclick = () => {

        const val = document.getElementById('search-input').value.trim().toLowerCase();

        if (!val) return;

        const found = nodes.find(n=>n.name.toLowerCase().includes(val));

        if (found && nodeMeshes[found.id]) {

          flyToNode(found.id);

          highlightNode(found.id, true);

          setTimeout(()=>highlightNode(null), 1800);

        }

      };


      // Decommission mode

      document.getElementById('decom-mode').addEventListener('change', e => {

        decomMode = e.target.checked;

        decomMainframe = null;

        decomImpacted = new Set();

        document.getElementById('decom-label').className = decomMode ? 'decom-on' : 'decom-off';

        updateNodeMaterials();

        document.getElementById('sidepanel').style.display = 'none';

      });


      animate();

    }


    let lastHoverNode = null;

    function onMouseMove(e) {

      const node = getIntersectedNode(e.clientX, e.clientY);

      if (node !== lastHoverNode) {

        highlightNode(node?.id);

        showTooltip(e.clientX, e.clientY, node);

        lastHoverNode = node;

      }

    }

    function onMouseClick(e) {

      const node = getIntersectedNode(e.clientX, e.clientY);

      if (!node) return;

      if (decomMode && node.type==='MAINFRAME') {

        decomMainframe = node.id;

        decomImpacted = computeDecomImpact(node.id);

        updateNodeMaterials();

        showSidepanel(node, decomImpacted.size);

      } else {

        showSidepanel(node);

      }

    }


    function animate(time=0) {

      requestAnimationFrame(animate);

      controls.update();

      animateEdges(time);

      renderer.render(scene, camera);

    }


    main();
